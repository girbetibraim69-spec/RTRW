Integrasi Auto-Save Google Sheets Induk

Menu ini hanya dapat dikonfigurasi oleh Super Admin / Admin Kelurahan.

Kode Apps Script (doPost) 
function doPost(e) {
  var payload = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(payload.module || "Sheet1");
  if (!sheet) sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d = payload.data;
  sheet.appendRow([new Date(), d.rt, d.rw, JSON.stringify(d), payload.user]);
  return ContentService.createTextOutput(JSON.stringify({status: "SUCCESS"})).setMimeType(ContentService.MimeType.JSON);
}
